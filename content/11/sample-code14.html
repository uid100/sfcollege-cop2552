/**
 * Chapter 14
 * Programming Challenge 7: Recursive Power Method
 */

public class PowDemo
{
   public static void main(String[] args)
   {
      double x = 2.0, y = 10.0;
      
      System.out.println(x + " raised to the power of " + y +
                         " is " + pow(x, y));
   }
   
   /**
    * pow method
    * This method returns the value of x raised to the
    * power of y.
    */
   
   public static double pow(double x, double y)
   {
      if (y > 0)
         return x * pow(x, y-1);
      else
         return 1;
   }
}
