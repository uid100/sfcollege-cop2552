Read user inputs from Java dialog boxes

        // Debbie Reid
        // NameDisplay
        // Prompt and display name using dialog boxes

        import javax.swing.JOptionPane;

        public class NameDisplay {
            public static void main(String[] args ) {
                //variables
                String firstName; //user input
                String lastName; //user input

                //get the user's first name
                firstName = JOptionPane.showInputDialog("What is your first name?");

                //get the user's last name
                lastName = JOptionPane.showInputDialog("What is your last name?");

                //display a greeting
                JOptionPane.showMessageDialog(null, "Hello " + firstName + " " + lastName);
                
                System.exit(0);
            }
        }

 

Console i/o

        // Eddie Van Halen
        // MilesPerGallon
        // Chapter 2: Programming Challenge 9: Miles-per-Gallon

        import java.util.Scanner; // Needed for the Scanner class

        public class MilesPerGallon
        {
            public static void main(String[] args)
            {
                double miles;        // Miles driven
                double gallons;      // Gallons of fuel
                double mpg;          // Miles-per-gallon
                
                // Create a Scanner object for keyboard input.
                Scanner keyboard = new Scanner(System.in);

                // Get the miles driven.
                System.out.print("Enter the miles driven: ");
                miles = keyboard.nextDouble();
                
                // Get the number of gallons of fuel.
                System.out.print("Enter the gallons of fuel used: ");
                gallons = keyboard.nextDouble();
                
                // Calculate the miles-per-gallon.
                mpg = miles / gallons;
                
                // Display the miles-per-gallon.
                System.out.printf("The miles-per-gallon is %,.0f", mpg);
            }
        }

 

Reminder: Java file names MUST match the class name!
