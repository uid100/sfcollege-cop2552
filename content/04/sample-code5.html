import java.util.Scanner;
import java.text.DecimalFormat;

/**
    * Chapter 5
    * Programming Challenge 4: Pennies for Pay
    */

public class PenniesForPay
{
    public static void main(String[] args)
    {
        int pennies;      // Penny accumulator
        int totalPay;     // Total pay accumulator
        int maxDays;      // Max number of days
        int day;          // Day counter

        // Create a Scanner object for keyboard input.
        Scanner keyboard = new Scanner(System.in);
    
        // Get the maximum number of days.
        System.out.print("For how many days will " +
                        "you work? ");
        maxDays = keyboard.nextInt();
    
        // Validate the input.
        while (maxDays < 1)
        {
            System.out.print("The number of days " +
                            "must be at least 1.\n" +
                            "Enter the number of days: ");
            maxDays = keyboard.nextInt();
        }
    
        // Initialize the day counter to day 1.
        day = 1;
    
        // Initialize the penny accumulator for
        // the first day at work.
        pennies = 1;
        
        // Initialize the total pay accumulator.
        totalPay = 0;
    
        // Display the report header.
        System.out.println("Day\t\tPennies Earned");
    
        // Display the income report.
        while (day <= maxDays)
        {
            // Display the day number and pennies earned.
            System.out.println(day + "\t\t" + pennies);
            
            // Accumulate the total pay.
            totalPay += pennies;
            
            // Increment for the next day.
            day++;
            
            // Double the number of pennies.
            pennies *= 2;
        }
    
        // Create a DecimalFormat object to format output.
        DecimalFormat dollar = new DecimalFormat("#,##0.00");
    
        // Display the total pay.
        System.out.println("Total pay: $" +
                            dollar.format(totalPay / 100.0));
    }
}

 

ReadWriteFile.java

//Best to import exactly what is needed instead of the entire library - for security purposes
//import java.io.*; - this would import everything we need plus lots of what we don't need.

import java.util.Scanner;
import java.io.File;
import java.io.PrintWriter;
import java.io.IOException;


public class ReadWriteFile {

    public static void main(String[] args) throws IOException  {
        PrintWriter outputFile = new PrintWriter("Original.txt");  //Note this will overwrite the file if it already exists.
        outputFile.println("Value 1");
        outputFile.println("Value 2");
        outputFile.println("Value 3");
        outputFile.close(); 
        outputFile = new PrintWriter("Duplicate.txt");
        

        // Open the file.
        File file = new File("Original.txt");
        Scanner inputFile = new Scanner(file);
        // Read until the end of the file.
        while (inputFile.hasNext())
        {
            String str = inputFile.nextLine();
            outputFile.println(str);
        }
        inputFile.close();// close the file when done.
        outputFile.close(); 
    }

}
